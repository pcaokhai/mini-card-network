# MCN-001 Monorepo scaffold and CI — Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans (small, tightly coupled story) to implement this plan task-by-task. Work in worktree `../mcn-worktrees/plat-001`, branch `feat/MCN-001-monorepo-ci`.

**Goal:** One repository with uniform `make` commands and a CI pipeline that lints/tests only what changed and enforces PR rules.

**Architecture:** Root `Makefile` delegates to per-service Makefiles (created by later stories) and to `infra/` and `contracts/`. GitHub Actions uses path filters; repository-level shell tests guard the Makefile and PR-size script.

**Tech stack:** GNU Make, Bash, GitHub Actions (`dorny/paths-filter`, `amannn/action-semantic-pull-request`).

- Compose files: `infra/docker-compose.yml` plus every `*/compose.yaml` (added by service stories), with `--project-directory .` so paths are repo-relative; `infra/scripts/post-up.sh` (MCN-002) runs after `up`.

```bash
#!/usr/bin/env bash
# MCN-001-AC1: every documented target exists, delegates, and propagates failures.
set -euo pipefail
cd "$(dirname "$0")/../.."
fail() { echo "FAIL: $*" >&2; exit 1; }

# Read make's rule database without running recipes ($(MAKE) lines would run even under -n).
db=$(make -pRrq -f Makefile : 2>/dev/null || true)
for t in help up down test lint fmt contracts e2e chaos pci-scan seed; do
  grep -qE "^${t}:" <<<"$db" || fail "target '$t' missing (MCN-001-AC1)"
done

make help | grep -q "contracts" || fail "help does not list targets"

tmp=$(mktemp -d)
trap 'rm -rf "$tmp"' EXIT
mkdir -p "$tmp/badsvc"
printf 'test:\n\t@exit 7\n' > "$tmp/badsvc/Makefile"
if make test SERVICES="$tmp/badsvc" >/dev/null 2>&1; then
  fail "root 'make test' must fail when a service fails (MCN-001-AC1)"
fi

mkdir -p "$tmp/goodsvc"
printf 'test:\n\t@echo ok-from-goodsvc\n' > "$tmp/goodsvc/Makefile"
make test SERVICES="$tmp/goodsvc" | grep -q ok-from-goodsvc || fail "delegation to service Makefile broken"

make test SERVICES="$tmp/does-not-exist" >/dev/null || fail "missing service dir must be skipped, not fail"
echo "PASS test_make_targets"
```

### Task 2: Root Makefile

**Files:** Create `Makefile`

**Step 1 — implement**

```makefile
# Root Makefile — delegates to services, infra and contracts. Owned by PLAT (see CODEOWNERS).
SHELL := /usr/bin/env bash
.SHELLFLAGS := -euo pipefail -c
SERVICES ?= issuer-jpos gateway-go web-next settlement
# Each service ships its own <service>/compose.yaml so parallel lanes never edit the same compose file.
COMPOSE_FILES := infra/docker-compose.yml $(wildcard */compose.yaml)
COMPOSE := docker compose --project-directory . $(addprefix -f ,$(COMPOSE_FILES)) --env-file .env

.PHONY: help up down test lint fmt contracts e2e chaos pci-scan seed

help: ## List targets
	@grep -E '^[a-z0-9-]+:.*## ' $(MAKEFILE_LIST) | awk -F':.*## ' '{printf "  %-10s %s\n", $$1, $$2}'

define delegate
	@for d in $(SERVICES); do \
	  if [ -f "$$d/Makefile" ]; then \
	    echo "==> $$d: make $(1)"; $(MAKE) --no-print-directory -C "$$d" $(1) || exit 1; \
	  else echo "--  $$d: not scaffolded yet, skipping $(1)"; fi; \
	done
endef

up: ## Start infra and services (docker compose, waits for health)
	@test -f infra/docker-compose.yml || { echo "infra not scaffolded (MCN-002)"; exit 1; }
	@test -f .env || cp .env.example .env
	$(COMPOSE) up -d --wait
	@if [ -x infra/scripts/post-up.sh ]; then infra/scripts/post-up.sh; fi

down: ## Stop everything; add ARGS=-v to delete volumes
	@test -f infra/docker-compose.yml || { echo "infra not scaffolded (MCN-002)"; exit 1; }
	$(COMPOSE) down $(ARGS)

test: ## Unit + integration tests of every service
	$(call delegate,test)

lint: ## Linters and format checks of every service
	$(call delegate,lint)

fmt: ## Auto-format every service
	$(call delegate,fmt)

seed: ## Load lab fixtures into every service database
	$(call delegate,seed)

contracts: ## Lint OpenAPI, validate schemas and ISO vectors, breaking-change check
	@test -f contracts/Makefile || { echo "contracts tooling not scaffolded (MCN-003)"; exit 1; }
	$(MAKE) --no-print-directory -C contracts check

e2e: ## Playwright journeys against the running stack
	$(MAKE) --no-print-directory -C web-next e2e

chaos: ## Chaos suite with ledger invariant (MCN-407)
	@echo "not implemented until MCN-407"; exit 2

pci-scan: ## Scan logs/fixtures/DB dump for card data (MCN-506)
	@echo "not implemented until MCN-506"; exit 2
```

### Task 5: PR size script

**Files:** Create `scripts/pr-size.sh`

```bash
#!/usr/bin/env bash
# Fails when a PR changes more than MAX_LINES lines outside generated/lock files.
set -euo pipefail
MAX_LINES="${MAX_LINES:-400}"
BASE_REF="${BASE_REF:-origin/main}"
EXCLUDE_REGEX='(/generated/|\.gen\.go$|_gen\.go$|\.pb\.go$|pnpm-lock\.yaml$|go\.sum$|\.lock$|/migrations/.*\.sql$)'
total=$(git diff --numstat "${BASE_REF}...HEAD" \
  | awk -v re="$EXCLUDE_REGEX" '$3 !~ re && $1 != "-" { s += $1 + $2 } END { print s + 0 }')
echo "changed lines (excluding generated): ${total} / ${MAX_LINES}"
if [ "${total}" -gt "${MAX_LINES}" ] && [ "${SIZE_EXCEPTION:-false}" != "true" ]; then
  echo "PR too large. Split the story or add the 'size-exception' label with a reason." >&2
  exit 1
fi
```

Run: `bash scripts/tests/test_pr_size.sh` → `PASS test_pr_size`. Commit: `ci(plat): PR size guard (MCN-001)`.

### Task 6: CI workflow

**Files:** Create `.github/workflows/ci.yml`

```yaml
name: ci
on:
  pull_request:
  push:
    branches: [main]
concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true
permissions:
  contents: read
  pull-requests: read

jobs:
  changes:
    runs-on: ubuntu-latest
    outputs:
      iss: ${{ steps.f.outputs.iss }}
      gw: ${{ steps.f.outputs.gw }}
      web: ${{ steps.f.outputs.web }}
      set: ${{ steps.f.outputs.set }}
    steps:
      - uses: actions/checkout@v4
      - id: f
        uses: dorny/paths-filter@v3
        with:
          filters: |
            iss: ['issuer-jpos/**', 'contracts/**']
            gw: ['gateway-go/**', 'contracts/**']
            web: ['web-next/**', 'contracts/**']
            set: ['settlement/**', 'contracts/**']

  repo-tests:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }
      - run: for t in scripts/tests/*.sh; do bash "$t"; done
      - if: github.event_name == 'pull_request'
        env:
          BASE_REF: origin/${{ github.base_ref }}
          SIZE_EXCEPTION: ${{ contains(github.event.pull_request.labels.*.name, 'size-exception') }}
        run: bash scripts/pr-size.sh

  contracts:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }
      - uses: actions/setup-python@v5
        with: { python-version: '3.12' }
      - uses: actions/setup-node@v4
        with: { node-version: 'lts/*' }
      - run: make contracts
        env:
          BASE_REF: origin/${{ github.base_ref || 'main' }}

  gateway:
    needs: changes
    if: needs.changes.outputs.gw == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-go@v5
        with: { go-version-file: gateway-go/go.mod }
      - run: make -C gateway-go lint test

  issuer:
    needs: changes
    if: needs.changes.outputs.iss == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with: { distribution: temurin, java-version: '21', cache: gradle }
      - run: make -C issuer-jpos lint test

  settlement:
    needs: changes
    if: needs.changes.outputs.set == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with: { distribution: temurin, java-version: '21', cache: gradle }
      - run: make -C settlement lint test

  web:
    needs: changes
    if: needs.changes.outputs.web == 'true'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with: { package_json_file: web-next/package.json }
      - uses: actions/setup-node@v4
        with: { node-version-file: web-next/.nvmrc, cache: pnpm, cache-dependency-path: web-next/pnpm-lock.yaml }
      - run: make -C web-next lint test
```

Note: service jobs only trigger once the service directory exists, because the path filter only matches when files under it change; the first PR creating a service therefore also creates its Makefile (MCN-004/005).
