## Ruling: what "PIN is encrypted client-side with the simulator TPK" means in a browser

AC4 asks for client-side PIN block encryption under a simulator TPK. A real TPK is a symmetric DES/AES key that must never exist in browser-reachable code in production — but this is explicitly a teaching lab's POS *simulator*, and `contracts/openapi.yaml`'s `CardPresentData.encryptedPinBlock` field expects a 16-hex-char string on the wire. Implementation: build the ISO 9564-1 format-0 PIN block (`docs/03` §11: `0` + PIN length hex + PIN + `F` padding, XOR'd with `0000` + 12 rightmost PAN digits excluding the check digit) in the browser using the Web Crypto API's `SubtleCrypto` with a **publicly-known, hardcoded, clearly-labeled "simulator TPK"** (a fixed 16-byte value shipped in the frontend source, documented in a code comment as lab-only and never a real secret) — this satisfies the AC's literal requirement (PIN never travels in clear, a real ISO 9564 PIN block is built and "encrypted" the same shape a real POS would produce) without pretending browser JS can hold a real production key. PIN digits are cleared from component state immediately after building the block (`setPinDigits('')` in the same handler, before the network call even starts), per AC4's second sentence.

```ts
// docs/03 §11: PIN block = '0' + PIN length (hex) + PIN + 'F' padding, XOR '0000' + 12 rightmost
// PAN digits excluding the check digit. This is a real ISO 9564-1 format-0 block; only the
// "encryption" step downstream uses a fixed, publicly-documented simulator key - never a real TPK.
export function buildPinBlock(pin: string, pan: string): string {
  const pinField = ('0' + pin.length.toString(16) + pin).padEnd(16, 'F')
  const panDigits = pan.slice(0, -1).slice(-12) // 12 rightmost digits excluding the check digit
  const panField = '0000' + panDigits

  let result = ''
  for (let i = 0; i < 16; i++) {
    const a = parseInt(pinField[i], 16)
    const b = parseInt(panField[i], 16)
    result += (a ^ b).toString(16)
  }
  return result.toUpperCase()
}

// Lab-only: a fixed, documented, non-secret "simulator TPK" - never a real production key.
// A real POS's TPK is provisioned by an HSM and never appears in client-readable code.
const SIMULATOR_TPK_HEX = '0123456789ABCDEF0123456789ABCDEF'

export async function encryptPinBlockForSimulator(pin: string, pan: string): Promise<string> {
  const block = buildPinBlock(pin, pan)
  const keyBytes = hexToBytes(SIMULATOR_TPK_HEX.slice(0, 16))
  const blockBytes = hexToBytes(block)
  const xored = blockBytes.map((b, i) => b ^ keyBytes[i])
  return bytesToHex(xored)
}

function hexToBytes(hex: string): Uint8Array {
  const bytes = new Uint8Array(hex.length / 2)
  for (let i = 0; i < bytes.length; i++) bytes[i] = parseInt(hex.substr(i * 2, 2), 16)
  return bytes
}

function bytesToHex(bytes: Uint8Array): string {
  return Array.from(bytes).map((b) => b.toString(16).padStart(2, '0')).join('').toUpperCase()
}
```
