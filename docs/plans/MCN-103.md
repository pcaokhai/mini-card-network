# MCN-103 Lab API — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans with superpowers:test-driven-development. Worktree `../mcn-worktrees/gw-103-lab-api`, branch `feat/MCN-103-lab-api`. Requires MCN-101 merged (codec) and MCN-005-GW merged (chi router, `internal/api`, `internal/obs` masking/logging already exist).

**Goal:** `POST /v1/lab/messages/decode`, `POST /v1/lab/messages/encode` and `GET /v1/lab/messages/samples`, validated against `contracts/openapi.yaml`'s `DecodedMessage`/`IsoField` schemas, with PAN masked in every response.

**Architecture:** A new `internal/lab` package wraps `internal/iso8583` (MCN-101) with an "easy name" glossary per DE and builds the `DecodedMessage` shape (segments for highlighting, ordered `IsoField` list). `internal/api` gains three chi routes under `/v1/lab/messages` that call `internal/lab` and marshal JSON. Response validation against the OpenAPI spec runs as a test-mode middleware (per `gateway-go/CLAUDE.md`'s test rules), not in production.

**Tech Stack:** Go 1.24, chi (already a dependency), `github.com/getkin/kin-openapi` (test-only, response validation), `encoding/json`.

**Spec:** `contracts/openapi.yaml` (`/v1/lab/messages/*`, `DecodedMessage`, `IsoField` schemas — already merged in the doc pack, no contract change needed for this story), `docs/03-iso8583-interface-spec.md` §3 (field table, used for easy names and DE 22/55/90 breakdown), `docs/06-user-stories.md` MCN-103, `gateway-go/CLAUDE.md`.

## Rulings

- **Easy names**: `packager-spec.yaml`'s `name` field (e.g. "Processing code", "Acquiring institution ID") is the `technicalName` for every DE. `easyName` is that same string for most fields (already learner-readable) with a small override map in `internal/lab/glossary.go` for the handful the story singles out: MTI (composed from its 4 digits, not a DE), DE 22 (entry mode code meaning, e.g. "051" → "Chip + PIN"), DE 55 (EMV, "ICC data (EMV tags)"), DE 90 (original data elements, "Reference to the original message"). This is a naming convenience, not an architectural decision — recorded here rather than escalated to a separate design session, and it is easy to extend the override map later without touching callers.
- **Segments** (for the raw-byte highlighting AC1/MCN-104-AC1 depends on) are computed by re-running the same tokenizer `Unpack` already performs, capturing each consumed slice's `(key, text)` instead of discarding it — see Task 1's `SegmentedUnpack`.
- The `raw` field on `IsoField` (present in the schema but not `required`) is the field's undecoded on-wire text (with length-prefix digits if any); omitted for MTI/bitmap since those aren't per-DE fields.

```go
package iso8583

// Segment is one consumed slice of the wire message, in read order, for UI highlighting (MCN-104).
type Segment struct {
	Key  string // "mti", "primaryBitmap", "secondaryBitmap", or the DE number as a string
	Text string // the exact on-wire text for this slice, including any length prefix
}

// UnpackSegmented behaves exactly like Unpack but additionally returns the ordered list of
// raw slices consumed, keyed by what each slice is — used by the Lab API to highlight bytes.
func UnpackSegmented(packed string) (string, map[int]string, []Segment, error) {
	pos := 0
	var segments []Segment
	take := func(count int, what string) (string, error) {
		if pos+count > len(packed) {
			return "", &CodecError{Code: "TRUNCATED", Detail: what + " needs " + itoa(count) + " chars at " + itoa(pos)}
		}
		s := packed[pos : pos+count]
		pos += count
		return s, nil
	}

	mti, err := take(4, "MTI")
	if err != nil {
		return "", nil, nil, err
	}
	if !digitsRe.MatchString(mti) {
		return "", nil, nil, &CodecError{Code: "INVALID_MTI", Detail: mti}
	}
	segments = append(segments, Segment{Key: "mti", Text: mti})

	primary, err := take(16, "primary bitmap")
	if err != nil {
		return "", nil, nil, err
	}
	if !hexRe.MatchString(primary) {
		return "", nil, nil, &CodecError{Code: "INVALID_BITMAP", Detail: primary}
	}
	segments = append(segments, Segment{Key: "primaryBitmap", Text: primary})
	bits, _ := strconvParseUint64(primary)
	var numbers []int
	for i := 0; i < 64; i++ {
		if bits&(1<<uint(63-i)) != 0 {
			numbers = append(numbers, i+1)
		}
	}
	if len(numbers) > 0 && numbers[0] == 1 {
		secondary, err := take(16, "secondary bitmap")
		if err != nil {
			return "", nil, nil, err
		}
		if !hexRe.MatchString(secondary) {
			return "", nil, nil, &CodecError{Code: "INVALID_BITMAP", Detail: secondary}
		}
		segments = append(segments, Segment{Key: "secondaryBitmap", Text: secondary})
		sbits, _ := strconvParseUint64(secondary)
		for i := 0; i < 64; i++ {
			if sbits&(1<<uint(63-i)) != 0 {
				numbers = append(numbers, 65+i)
			}
		}
	}

	fields := make(map[int]string, len(numbers))
	for _, n := range numbers {
		if n == 1 {
			continue
		}
		spec, ok := Fields[n]
		if !ok {
			return "", nil, nil, &CodecError{Code: "UNKNOWN_FIELD", DE: n}
		}
		var lenPrefix string
		var size int
		if spec.Prefix != "" {
			digits := 2
			if spec.Prefix == "LLL" {
				digits = 3
			}
			rawLen, err := take(digits, "DE "+itoa(n)+" length")
			if err != nil {
				return "", nil, nil, err
			}
			if !digitsRe.MatchString(rawLen) {
				return "", nil, nil, &CodecError{Code: "INVALID_LENGTH", DE: n, Detail: "prefix '" + rawLen + "'"}
			}
			lenPrefix = rawLen
			size, _ = strconvAtoi(rawLen)
			if size > spec.Length {
				return "", nil, nil, &CodecError{Code: "INVALID_LENGTH", DE: n, Detail: "max " + itoa(spec.Length) + ", got " + itoa(size)}
			}
		} else {
			size = spec.Length
		}
		width := size
		if spec.Type == "b" {
			width = size * 2
		}
		value, err := take(width, "DE "+itoa(n))
		if err != nil {
			return "", nil, nil, err
		}
		if err := checkValue(spec, value); err != nil {
			return "", nil, nil, err
		}
		fields[n] = value
		segments = append(segments, Segment{Key: itoa(n), Text: lenPrefix + value})
	}
	if pos != len(packed) {
		return "", nil, nil, &CodecError{Code: "TRAILING_DATA", Detail: itoa(len(packed)-pos) + " chars after last field"}
	}
	return mti, fields, segments, nil
}
```

```go
package lab

import (
	"sort"
	"strconv"

	"github.com/mcn/gateway-go/internal/iso8583"
	"github.com/mcn/gateway-go/internal/obs"
)

// Field is one decoded data element, ready for the Lab UI (contracts/openapi.yaml IsoField).
type Field struct {
	DE            string
	EasyName      string
	TechnicalName string
	Format        string
	Value         string
	Raw           string
}

// Segment mirrors iso8583.Segment for JSON marshaling under this package's own name.
type Segment = iso8583.Segment

// Decoded is the Lab API's response shape (contracts/openapi.yaml DecodedMessage).
type Decoded struct {
	MTI             string
	PrimaryBitmap   string
	SecondaryBitmap *string
	Segments        []Segment
	Fields          []Field
}

func buildFields(fields map[int]string, segmentsByDE map[int]string) []Field {
	numbers := make([]int, 0, len(fields))
	for n := range fields {
		numbers = append(numbers, n)
	}
	sort.Ints(numbers)
	out := make([]Field, 0, len(numbers))
	for _, n := range numbers {
		spec := iso8583.Fields[n]
		value := fields[n]
		if n == 2 {
			value = obs.MaskPAN(value)
		}
		out = append(out, Field{
			DE:            strconv.Itoa(n),
			EasyName:      easyName(n, spec.Name()),
			TechnicalName: spec.Name(),
			Format:        spec.Type,
			Value:         value,
			Raw:           segmentsByDE[n],
		})
	}
	return out
}

// Decode unpacks a raw wire message into the Lab API's DecodedMessage shape.
func Decode(raw string) (Decoded, error) {
	mti, fields, segments, err := iso8583.UnpackSegmented(raw)
	if err != nil {
		return Decoded{}, err
	}
	return assemble(mti, fields, segments), nil
}

// Encode packs fields (keyed by DE number as a string, per the OpenAPI request body) and
// returns the same DecodedMessage shape as Decode, so the UI can show the breakdown either way.
func Encode(mti string, fields map[string]string) (Decoded, error) {
	numeric := make(map[int]string, len(fields))
	for k, v := range fields {
		n, convErr := strconv.Atoi(k)
		if convErr != nil {
			return Decoded{}, &iso8583.CodecError{Code: "UNKNOWN_FIELD", Detail: "DE key not numeric: " + k}
		}
		numeric[n] = v
	}
	packed, err := iso8583.Pack(mti, numeric)
	if err != nil {
		return Decoded{}, err
	}
	return Decode(packed)
}

func assemble(mti string, fields map[int]string, segments []Segment) Decoded {
	segByDE := make(map[int]string, len(segments))
	var primary, secondary string
	for _, s := range segments {
		switch s.Key {
		case "primaryBitmap":
			primary = s.Text
		case "secondaryBitmap":
			secondary = s.Text
		case "mti":
			// nothing to capture per-field
		default:
			n, _ := strconv.Atoi(s.Key)
			segByDE[n] = s.Text
		}
	}
	d := Decoded{MTI: mti, PrimaryBitmap: primary, Segments: segments, Fields: buildFields(fields, segByDE)}
	if secondary != "" {
		d.SecondaryBitmap = &secondary
	}
	return d
}
```

```go
package api

import (
	"encoding/json"
	"net/http"

	"github.com/go-chi/chi/v5"

	"github.com/mcn/gateway-go/internal/lab"
)

// MountLab registers the Message Lab routes (contracts/openapi.yaml, tag "lab").
func MountLab(r chi.Router) {
	r.Post("/v1/lab/messages/decode", handleDecode)
	r.Post("/v1/lab/messages/encode", handleEncode)
	r.Get("/v1/lab/messages/samples", handleSamples)
}

func handleDecode(w http.ResponseWriter, req *http.Request) {
	var body struct {
		Raw string `json:"raw"`
	}
	if err := json.NewDecoder(req.Body).Decode(&body); err != nil {
		problem(w, http.StatusBadRequest, "invalid-request", err.Error())
		return
	}
	d, err := lab.Decode(body.Raw)
	if err != nil {
		problem(w, http.StatusBadRequest, "invalid-message", err.Error())
		return
	}
	writeJSONBody(w, http.StatusOK, d)
}

func handleEncode(w http.ResponseWriter, req *http.Request) {
	var body struct {
		MTI    string            `json:"mti"`
		Fields map[string]string `json:"fields"`
	}
	if err := json.NewDecoder(req.Body).Decode(&body); err != nil {
		problem(w, http.StatusBadRequest, "invalid-request", err.Error())
		return
	}
	d, err := lab.Encode(body.MTI, body.Fields)
	if err != nil {
		problem(w, http.StatusBadRequest, "invalid-message", err.Error())
		return
	}
	writeJSONBody(w, http.StatusOK, d)
}

func handleSamples(w http.ResponseWriter, _ *http.Request) {
	writeJSONBody(w, http.StatusOK, lab.Samples)
}

func writeJSONBody(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}

// problem writes an RFC 7807 problem+json body (docs/04 §2 error convention).
func problem(w http.ResponseWriter, status int, typ, detail string) {
	w.Header().Set("Content-Type", "application/problem+json")
```

```go
package api

import (
	"bytes"
	"context"
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"testing"

	"github.com/getkin/kin-openapi/openapi3"

	"github.com/getkin/kin-openapi/routers/gorillamux"
	"github.com/go-chi/chi/v5"
)

func loadSpec(t *testing.T) *openapi3.T {
	t.Helper()
	loader := openapi3.NewLoader()
	doc, err := loader.LoadFromFile("../../../contracts/openapi.yaml")
	if err != nil {
		t.Fatalf("load spec: %v", err)
	}
	if err := doc.Validate(loader.Context); err != nil {
		t.Fatalf("spec invalid: %v", err)
	}
	return doc
}

func TestLabResponses_matchOpenAPISpec__MCN_103_AC4(t *testing.T) {
	doc := loadSpec(t)
	router, err := gorillamux.NewRouter(doc)
	if err != nil {
		t.Fatalf("build router: %v", err)
	}

	r := chi.NewRouter()
	MountLab(r)

	cases := []*http.Request{
		httptest.NewRequest(http.MethodGet, "/v1/lab/messages/samples", nil),
		httptest.NewRequest(http.MethodPost, "/v1/lab/messages/decode",
			bytes.NewReader(mustJSON(map[string]string{"raw": "0800822000000000000004000000000000000921073300000200301"}))),
	}
	for _, req := range cases {
		req.Header.Set("Content-Type", "application/json")
		route, pathParams, err := router.FindRoute(req)
		if err != nil {
			t.Fatalf("find route for %s: %v", req.URL.Path, err)
		}
		rec := httptest.NewRecorder()
		r.ServeHTTP(rec, req)

		responseValidationInput := &openapi3filter.ResponseValidationInput{
			RequestValidationInput: &openapi3filter.RequestValidationInput{Request: req, PathParams: pathParams, Route: route},
			Status:                 rec.Code,
			Header:                 rec.Header(),
		}
		responseValidationInput.SetBodyBytes(rec.Body.Bytes())
		if err := openapi3filter.ValidateResponse(context.Background(), responseValidationInput); err != nil {
			t.Fatalf("response for %s does not match spec: %v", req.URL.Path, err)
		}
	}
}

func mustJSON(v any) []byte {
	b, _ := json.Marshal(v)
	return b
}
```
