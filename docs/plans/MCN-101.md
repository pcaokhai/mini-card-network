# MCN-101 ISO 8583 codec (Go) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans with superpowers:test-driven-development. Worktree `../mcn-worktrees/gw-101-codec`, branch `feat/MCN-101-iso8583-codec`. Requires MCN-003 merged (it is — `contracts/` already has `packager-spec.yaml` and golden/invalid vectors).

**Goal:** A pure, generated-from-spec ISO 8583 (MCN-87A) codec in `gateway-go/internal/iso8583` that packs/unpacks every field type in `packager-spec.yaml`, round-trips every golden vector byte-identical, returns typed errors with the DE number for every invalid vector, survives 60s of fuzzing, and is cross-checked against `moov-io/iso8583` in a test-only dependency.

**Architecture:** `internal/iso8583/spec_gen.go` is generated from `contracts/iso8583/packager-spec.yaml` by a small generator (`internal/iso8583/gen/main.go`, run via `go generate`) — this is the exact same source of truth MCN-102 generates `iso87ascii.xml` from (ADR-003). `codec.go` holds `Pack`/`Unpack` as pure functions over the generated field table; no I/O, no globals. This is a direct, verified-by-existing-tests port of `contracts/scripts/mcn87a.py` (MCN-003's Python reference codec, whose behavior is already golden-vector-verified) — same algorithm, same error semantics, Go idioms.

**Tech Stack:** Go 1.24 (matches `gateway-go/go.mod`), `gopkg.in/yaml.v3` (generator only, not a runtime dependency of the codec), `github.com/stretchr/testify`, `github.com/moov-io/iso8583` (test-only, for AC4's cross-check).

**Spec:** `docs/03-iso8583-interface-spec.md` §2–3 (encoding, field table), `contracts/iso8583/packager-spec.yaml`, `contracts/iso8583/vectors/*.json` (+ `invalid/*.json`), `docs/06-user-stories.md` MCN-101, `docs/adr/ADR-003-own-iso-codec-go.md`, `gateway-go/CLAUDE.md` (layout, Go rules, test rules).

## Global Constraints

- No I/O, no global mutable state in `internal/iso8583` — pure functions only (`gateway-go/CLAUDE.md`).
- Money stays out of this package; DE 4 is packed/unpacked as its raw ASCII string, conversion to `int64` minor units happens in `internal/txn` later.
- PAN (DE 2), PIN block (DE 52), key material (DE 48), EMV data (DE 55) are never logged by this package; masking is a caller concern (`internal/obs`, MCN-103). This package only packs/unpacks.
- `go test -race` must pass (project-wide rule); the codec has no concurrency of its own, but the fuzz target must not introduce shared mutable state across fuzz workers.
- Golden vectors are the source of truth for correctness — if a vector and a hand-written test disagree, the vector wins (docs/03: "this document wins").

```go
package main

import (
	"fmt"
	"log"
	"os"
	"sort"
	"strconv"

	"gopkg.in/yaml.v3"
)

type rawField struct {
	Name      string `yaml:"name"`
	Type      string `yaml:"type"`
	Length    int    `yaml:"length"`
	Prefix    string `yaml:"prefix"`
	Sensitive string `yaml:"sensitive"`
}

type rawSpec struct {
	Fields map[int]rawField `yaml:"fields"`
}

func main() {
	data, err := os.ReadFile("../contracts/iso8583/packager-spec.yaml")
	if err != nil {
		log.Fatal(err)
	}
	var spec rawSpec
	if err := yaml.Unmarshal(data, &spec); err != nil {
		log.Fatal(err)
	}
	numbers := make([]int, 0, len(spec.Fields))
	for n := range spec.Fields {
		numbers = append(numbers, n)
	}
	sort.Ints(numbers)

	out, err := os.Create("spec_gen.go")
	if err != nil {
		log.Fatal(err)
	}
	defer out.Close()

	fmt.Fprintln(out, "// Code generated from contracts/iso8583/packager-spec.yaml by internal/iso8583/gen. DO NOT EDIT.")
	fmt.Fprintln(out, "package iso8583")
	fmt.Fprintln(out)
	fmt.Fprintln(out, "// Fields is the MCN-87A field table (docs/03 §3).")
	fmt.Fprintln(out, "var Fields = map[int]FieldSpec{")
	for _, n := range numbers {
		f := spec.Fields[n]
		fmt.Fprintf(out, "\t%d: {Number: %d, Type: %q, Length: %d, Prefix: %q}, // %s\n",
			n, n, f.Type, f.Length, f.Prefix, f.Name)
```

```go
package iso8583

import "errors"

// CodecError is a typed pack/unpack failure carrying the offending DE number (0 = MTI/bitmap level).
type CodecError struct {
	Code   string // INVALID_MTI, INVALID_BITMAP, UNKNOWN_FIELD, INVALID_LENGTH, INVALID_VALUE, TRUNCATED, TRAILING_DATA
	DE     int
	Detail string
}

func (e *CodecError) Error() string {
	if e.DE > 0 {
		return e.Code + ": DE " + itoa(e.DE) + ": " + e.Detail
	}
	return e.Code + ": " + e.Detail
}

// AsCodecError is a thin errors.As wrapper so callers don't need to import this package's error type name twice.
func AsCodecError(err error, target **CodecError) bool {
	return errors.As(err, target)
}

func itoa(n int) string {
	if n == 0 {
		return "0"
	}
	neg := n < 0
	if neg {
		n = -n
	}
	var buf [20]byte
	i := len(buf)
	for n > 0 {
		i--
		buf[i] = byte('0' + n%10)
		n /= 10
	}
	if neg {
		i--
		buf[i] = '-'
	}
	return string(buf[i:])
}
```

```go
package iso8583

import (
	"regexp"
	"strconv"
	"strings"
)

var hexRe = regexp.MustCompile(`^[0-9A-F]*$`)
var digitsRe = regexp.MustCompile(`^[0-9]*$`)

func dataLen(f FieldSpec, value string) int {
	if f.Type == "b" {
		return len(value) / 2
	}
	return len(value)
}

func checkValue(f FieldSpec, value string) error {
	if f.Type == "b" && (len(value)%2 != 0 || !hexRe.MatchString(value)) {
		return &CodecError{Code: "INVALID_VALUE", DE: f.Number, Detail: "must be uppercase hex"}
	}
	if f.Type == "n" && !digitsRe.MatchString(value) {
		return &CodecError{Code: "INVALID_VALUE", DE: f.Number, Detail: "must be numeric"}
	}
	size := dataLen(f, value)
	if f.Prefix == "" && size != f.Length {
		return &CodecError{Code: "INVALID_LENGTH", DE: f.Number, Detail: "fixed length " + itoa(f.Length) + ", got " + itoa(size)}
	}
	if f.Prefix != "" && size > f.Length {
		return &CodecError{Code: "INVALID_LENGTH", DE: f.Number, Detail: "max " + itoa(f.Length) + ", got " + itoa(size)}
	}
	return nil
}

func bitmapHex(numbers map[int]bool, page int) string {
	var value uint64
	for i := 0; i < 64; i++ {
		if numbers[page*64+i+1] {
			value |= 1 << uint(63-i)
		}
	}
	return strings.ToUpper(pad16(strconv.FormatUint(value, 16)))
}

func pad16(s string) string {
	for len(s) < 16 {
		s = "0" + s
	}
	return s
}

// Pack builds a wire message for mti with the given DE number -> value map. DE 1 (secondary
// bitmap) is derived automatically and must not be set explicitly.
func Pack(mti string, fields map[int]string) (string, error) {
	if len(mti) != 4 || !digitsRe.MatchString(mti) {
		return "", &CodecError{Code: "INVALID_MTI", Detail: mti}
	}
	if _, has1 := fields[1]; has1 {
		return "", &CodecError{Code: "UNKNOWN_FIELD", Detail: "DE 1 is derived, do not set it"}
	}
	present := make(map[int]bool, len(fields))
	secondary := false
	for n := range fields {
		present[n] = true
		if n > 64 {
			secondary = true
		}
	}
	if secondary {
		present[1] = true
	}

	numbers := make([]int, 0, len(fields))
	for n := range fields {
		numbers = append(numbers, n)
	}
	sortInts(numbers)

	var b strings.Builder
	b.WriteString(mti)
	b.WriteString(bitmapHex(present, 0))
	if secondary {
		b.WriteString(bitmapHex(present, 1))
	}
	for _, n := range numbers {
		spec, ok := Fields[n]
		if !ok {
			return "", &CodecError{Code: "UNKNOWN_FIELD", DE: n}
		}
		v := fields[n]
		if err := checkValue(spec, v); err != nil {
			return "", err
		}
		switch spec.Prefix {
		case "LL":
			b.WriteString(pad2(dataLen(spec, v)))
		case "LLL":
			b.WriteString(pad3(dataLen(spec, v)))
		}
		b.WriteString(v)
	}
	return b.String(), nil
}

func pad2(n int) string { return padN(n, 2) }
func pad3(n int) string { return padN(n, 3) }
func padN(n, width int) string {
	s := strconv.Itoa(n)
	for len(s) < width {
		s = "0" + s
	}
	return s
}

func sortInts(a []int) {
	for i := 1; i < len(a); i++ {
		for j := i; j > 0 && a[j-1] > a[j]; j-- {
			a[j-1], a[j] = a[j], a[j-1]
		}
	}
}

// Unpack parses a wire message into its MTI and DE number -> value map.
func Unpack(packed string) (string, map[int]string, error) {
	pos := 0
	take := func(count int, what string) (string, error) {
		if pos+count > len(packed) {
			return "", &CodecError{Code: "TRUNCATED", Detail: what + " needs " + itoa(count) + " chars at " + itoa(pos)}
		}
		s := packed[pos : pos+count]
		pos += count
		return s, nil
	}

	mti, err := take(4, "MTI")
	if err != nil {
		return "", nil, err
	}
	if !digitsRe.MatchString(mti) {
		return "", nil, &CodecError{Code: "INVALID_MTI", Detail: mti}
	}
	primary, err := take(16, "primary bitmap")
	if err != nil {
		return "", nil, err
	}
	if !hexRe.MatchString(primary) {
		return "", nil, &CodecError{Code: "INVALID_BITMAP", Detail: primary}
	}
	bits, _ := strconv.ParseUint(primary, 16, 64)
	var numbers []int
	for i := 0; i < 64; i++ {
		if bits&(1<<uint(63-i)) != 0 {
			numbers = append(numbers, i+1)
		}
	}
	if len(numbers) > 0 && numbers[0] == 1 {
		secondary, err := take(16, "secondary bitmap")
		if err != nil {
			return "", nil, err
		}
		if !hexRe.MatchString(secondary) {
			return "", nil, &CodecError{Code: "INVALID_BITMAP", Detail: secondary}
		}
		sbits, _ := strconv.ParseUint(secondary, 16, 64)
		for i := 0; i < 64; i++ {
			if sbits&(1<<uint(63-i)) != 0 {
				numbers = append(numbers, 65+i)
			}
		}
	}

	fields := make(map[int]string, len(numbers))
	for _, n := range numbers {
		if n == 1 {
			continue
		}
		spec, ok := Fields[n]
		if !ok {
			return "", nil, &CodecError{Code: "UNKNOWN_FIELD", DE: n}
		}
		var size int
		if spec.Prefix != "" {
			digits := 2
			if spec.Prefix == "LLL" {
				digits = 3
			}
			rawLen, err := take(digits, "DE "+itoa(n)+" length")
			if err != nil {
				return "", nil, err
			}
			if !digitsRe.MatchString(rawLen) {
				return "", nil, &CodecError{Code: "INVALID_LENGTH", DE: n, Detail: "prefix '" + rawLen + "'"}
			}
			size, _ = strconv.Atoi(rawLen)
			if size > spec.Length {
				return "", nil, &CodecError{Code: "INVALID_LENGTH", DE: n, Detail: "max " + itoa(spec.Length) + ", got " + itoa(size)}
			}
		} else {
			size = spec.Length
		}
		width := size
		if spec.Type == "b" {
			width = size * 2
		}
		value, err := take(width, "DE "+itoa(n))
		if err != nil {
			return "", nil, err
		}
		if err := checkValue(spec, value); err != nil {
			return "", nil, err
		}
		fields[n] = value
	}
	if pos != len(packed) {
		return "", nil, &CodecError{Code: "TRAILING_DATA", Detail: itoa(len(packed)-pos) + " chars after last field"}
	}
	return mti, fields, nil
}
```

```go
type goldenVector struct {

type invalidVector struct {

	Packed        string `json:"packed"`

func loadJSON[T any](t *testing.T, path string) T {
	t.Helper()
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	var v T
	if err := json.Unmarshal(data, &v); err != nil {
		t.Fatal(err)
	}
	return v
}

func TestGoldenVectors_roundTripByteIdentical__MCN_101_AC1(t *testing.T) {
	paths, _ := filepath.Glob("../../../contracts/iso8583/vectors/*.json")
	if len(paths) == 0 {
		t.Fatal("no golden vectors found")
	}
	for _, p := range paths {
		p := p
		t.Run(filepath.Base(p), func(t *testing.T) {
			v := loadJSON[goldenVector](t, p)
			fields := make(map[int]string, len(v.Fields))
			for k, val := range v.Fields {
				var n int
				for _, c := range k {
					n = n*10 + int(c-'0')
				}
				fields[n] = val
			}
			packed, err := Pack(v.MTI, fields)
			if err != nil {
				t.Fatalf("Pack: %v", err)
			}
			if packed != v.Packed {
				t.Fatalf("pack mismatch\n got: %s\nwant: %s", packed, v.Packed)
			}
			mti, back, err := Unpack(v.Packed)
			if err != nil {
				t.Fatalf("Unpack: %v", err)
			}
			if mti != v.MTI {
				t.Fatalf("mti mismatch: got %s want %s", mti, v.MTI)
			}
			for k, want := range v.Fields {
				n := 0
				for _, c := range k {
					n = n*10 + int(c-'0')
				}
				if back[n] != want {
					t.Fatalf("DE %s mismatch: got %s want %s", k, back[n], want)
				}
			}
		})
	}
}
```

```go
// MessageSpec, so both libraries are driven by the identical field table — this is a
// cross-check of the *codec logic*, not of two independently-chosen specs.
func buildMoovSpec() *moov.MessageSpec {
	spec := &moov.MessageSpec{Fields: map[int]field.Field{0: field.NewString(&field.Spec{Length: 4, Description: "MTI", Enc: encoding.ASCII, Pref: prefix.ASCII.Fixed})}}
	for n, f := range Fields {
		var enc encoding.Encoder = encoding.ASCII
		var pad padding.Padder = padding.Left('0')
		if f.Type == "an" || f.Type == "ans" {
			pad = padding.Right(' ')
		}
		if f.Type == "b" {
			enc = encoding.HexToASCII
		}
		var pref prefix.Prefixer
		switch f.Prefix {
		case "LL":
			pref = prefix.ASCII.LL
		case "LLL":
			pref = prefix.ASCII.LLL
		default:
			pref = prefix.ASCII.Fixed
		}
		spec.Fields[n] = field.NewString(&field.Spec{Length: f.Length, Description: itoa(n), Enc: enc, Pref: pref, Pad: pad})
	}
	return spec
}
```
